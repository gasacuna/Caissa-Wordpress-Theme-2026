# =========================================================================
# CACHÉ DE LOS ASSETS DEL TEMA - Caissa
#
# Va en el mismo .htaccess de la raíz que el bloque de redirects, también
# ARRIBA de "# BEGIN WordPress". Los dos bloques son independientes.
#
# QUÉ ARREGLA, MEDIDO EL 07/09/2026
#
# Hoy el servidor manda "Cache-Control: public, max-age=604800" (7 días) para
# el CSS, el JS, las fuentes y las imágenes del tema. Lighthouse marca como
# defecto cualquier valor por debajo de un año en un recurso versionado, y en
# una visita repetida el navegador revalida en vez de servir de disco.
#
# POR QUÉ UN AÑO ES SEGURO ACÁ, Y NO LO SERÍA EN CUALQUIER TEMA
#
# El tema encola TODOS sus assets con "?ver=CAISSA_VER", y esa constante se
# reescribe en cada build (hoy 1.15). Al subir una versión nueva cambia la
# query de cada archivo, así que el navegador pide una URL distinta y no hay
# forma de que sirva la vieja. Un año de caché sobre una URL que cambia con
# cada deploy no puede quedar rancio.
#
# ⚠️ La única excepción son las FUENTES, que el tema referencia desde el CSS
# sin "?ver=". Igual llevan un año: son archivos que no cambian nunca, y si
# alguna vez se reemplaza una hay que renombrarla.
#
# LO QUE ESTE BLOQUE NO ARREGLA
#
# Nada del servidor. La medición dice que el HTML se sirve con
# "x-powered-by: PHP" en cada petición, o sea que NO hay caché de página, y el
# TTFB queda entre 288 y 542 ms. Eso no se resuelve desde el .htaccess: se
# resuelve activando la caché de LiteSpeed en el panel. Es el lever más grande
# que queda y no es de este archivo.
# =========================================================================

<IfModule mod_expires.c>
ExpiresActive On

# Un año para lo que el tema versiona con ?ver= y para las fuentes.
ExpiresByType text/css                  "access plus 1 year"
ExpiresByType application/javascript    "access plus 1 year"
ExpiresByType text/javascript           "access plus 1 year"
ExpiresByType font/woff2                "access plus 1 year"
ExpiresByType image/webp                "access plus 1 year"
ExpiresByType image/svg+xml             "access plus 1 year"
ExpiresByType image/png                 "access plus 1 year"
ExpiresByType image/jpeg                "access plus 1 year"

# El HTML NO se cachea en el navegador: cada visita tiene que ver la versión
# publicada. La caché del HTML es la del servidor (LiteSpeed), no la del
# cliente, y son dos cosas distintas.
ExpiresByType text/html                 "access plus 0 seconds"
</IfModule>

# mod_expires escribe Expires y max-age. immutable no lo pone, y es el que le
# dice al navegador que no revalide ni con un refresh: se puede afirmar sin
# riesgo justamente porque la URL lleva ?ver=.
<IfModule mod_headers.c>
  <FilesMatch "\.(css|js|woff2|webp|svg|png|jpe?g|avif)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  # El HTML, explícito, para que ningún proxy intermedio se lo guarde.
  <FilesMatch "\.(html|php)$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
</IfModule>

# =========================================================================
# COMPRESIÓN: NO HACE FALTA. Verificado el 07/09/2026: el servidor ya manda
# "Content-Encoding: br" en el HTML, el CSS y el JS. Un bloque de mod_deflate
# acá sería redundante y podría pisar el Brotli con gzip, que comprime peor.
# =========================================================================


# =========================================================================
# CABECERAS DE SEGURIDAD
#
# Esto sube el **77 de Prácticas recomendadas**, no el rendimiento. De los
# nueve ítems que Lighthouse marca en esa categoría, cinco son cabeceras que
# el servidor no manda, y tres de esos cinco se resuelven acá sin ningún
# riesgo. Van juntas en este archivo porque es el mismo `.htaccess`.
# =========================================================================

<IfModule mod_headers.c>

  # HSTS: obliga a https durante un año. Requisito para el ítem "Usa una
  # política de HSTS sólida".
  #
  # ⚠️ Antes de activarlo hay que estar seguro de que TODO el sitio y todos
  # los subdominios andan por https. Con includeSubDomains, un subdominio que
  # sólo exista en http queda inaccesible para quien ya visitó el sitio, y el
  # navegador recuerda la orden durante un año: no se puede revertir sacando
  # la cabecera. El staging (demo.caissa.digital) es un subdominio y hoy anda
  # por https, así que está bien; si mañana aparece uno que no, hay que sacar
  # includeSubDomains ANTES.
  #
  # "preload" NO se pone a propósito: eso pide entrar en la lista que viene
  # compilada en los navegadores, y salir de esa lista lleva meses.
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"

  # Clickjacking. Resuelve "Mitigar el clickjacking con XFO o CSP".
  # SAMEORIGIN y no DENY: el propio sitio embebe iframes (el mapa de Neuquén,
  # el Typeform, los videos de YouTube) y DENY rompería la vista previa del
  # editor de WordPress.
  Header always set X-Frame-Options "SAMEORIGIN"

  # Aislamiento de origen. Resuelve "Asegúrate de que el aislamiento de origen
  # sea adecuado con COOP".
  #
  # same-origin-allow-popups y NO same-origin: con same-origin se rompen los
  # flujos que abren una ventana y esperan respuesta (los login social, y
  # cualquier popup de pago). allow-popups deja pasar esos y cumple igual.
  Header always set Cross-Origin-Opener-Policy "same-origin-allow-popups"

  # Estas dos no las pide Lighthouse pero son gratis y estándar.
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"

</IfModule>

#

# Types. Los dos exigen declarar de dónde puede venir cada script, y este
# sitio carga GTM, GA4, Google Ads, el píxel de Meta, LinkedIn, YouTube,
# Typeform y Google Maps. Una CSP escrita a ciegas rompe el seguimiento en
# silencio: los scripts dejan de ejecutarse y no hay error visible en la
# página, sólo datos que no llegan.
